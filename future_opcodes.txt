enum finvoke_operation {
	FINV_ASIN	= 0x00,
	FINV_ACOS	= 0x01,
	FINV_ATAN	= 0x02,
	FINV_ATAN2	= 0x03,
	FINV_EXP	= 0x04,
	FINV_LOG	= 0x05,
	FINV_LOG2	= 0x06,
	FINV_LOG10	= 0x07,
	FINV_TRUNC	= 0x08,
	FINV_ROUND	= 0x09
};


/*
MNVK argument contracts:

CREATE:
	r0 = matrix ID
	r1 = rows
	r2 = columns
	Создать матрицу, заполненную нулями.

SET:
	r0 = matrix ID
	r1 = row
	r2 = column
	r3 = source float-slot key

GET:
	r0 = matrix ID
	r1 = row
	r2 = column
	r3 = destination float-slot key

COPY_ELEM / SWAP_ELEM:
	r0 = destination matrix ID
	r1 = destination row
	r2 = destination column
	r3 = source matrix ID
	r4 = source row
	r5 = source column

ADD / SUB / MUL:
	r0 = destination matrix ID
	r1 = matrix A ID
	r2 = matrix B ID

SCALE:
	r0 = destination matrix ID
	r1 = source matrix ID
	r2 = scalar float-slot key

Common:
	out = error code; 0 means success.
	Argument registers remain unchanged.
	Destination matrices must already exist and have suitable dimensions.
*/
enum minvoke_operation {
	MINV_CREATE	= 0x00,
	MINV_SET		= 0x01,
	MINV_GET		= 0x02,
	MINV_COPY_ELEM	= 0x03,
	MINV_SWAP_ELEM	= 0x04,

	MINV_ADD		= 0x05,
	MINV_SUB		= 0x06,
	MINV_SCALE		= 0x07,
	MINV_MUL		= 0x08
};
